   140→/**
   141→ * Safely deduct inventory with atomic operation to prevent race conditions
   142→ * Uses optimistic concurrency control for SQLite
   143→ */
   144→async function safeInventoryDeduct(
   145→  tx: any,
   146→  branchId: string,
   147→  ingredientId: string,
   148→  quantityToDeduct: number,
   149→  orderId: string,
   150→  createdBy: string,
   151→  ingredientName: string
   152→) {
   153→  const quantityToDeductAbs = Math.abs(quantityToDeduct);
   154→
   155→  // Check current stock first
   156→  const inventory = await tx.branchInventory.findUnique({
   157→    where: {
   158→      branchId_ingredientId: {
   159→        branchId,
   160→        ingredientId,
   161→      },
   162→    },
   163→  });
   164→
   165→  const currentStock = inventory?.currentStock || 0;
   166→
   167→  // Allow negative stock - just deduct the quantity
   168→  // This allows orders to proceed even when stock is insufficient
   169→
   170→  // Update inventory (or create if doesn't exist)
   171→  const stockBefore = currentStock;
   172→  const stockAfter = currentStock - quantityToDeductAbs;
   173→
   174→  // Use upsert to handle case where inventory record doesn't exist yet
   175→  await tx.branchInventory.upsert({
   176→    where: {
   177→      branchId_ingredientId: {
   178→        branchId,
   179→        ingredientId,
   180→      },
   181→    },
   182→    create: {
   183→      branchId,
   184→      ingredientId,
   185→      currentStock: stockAfter,
   186→      reservedStock: 0,
   187→      lastModifiedAt: new Date(),
   188→    },
   189→    update: {
   190→      currentStock: stockAfter,
   191→      lastModifiedAt: new Date(),
   192→    },
   193→  });
   194→
   195→  // Create inventory transaction record
   196→  await tx.inventoryTransaction.create({
   197→    data: {
   198→      branchId,
   199→      ingredientId,
   200→      transactionType: 'SALE',
   201→      quantityChange: -quantityToDeductAbs,
   202→      stockBefore,
   203→      stockAfter,
   204→      orderId,
   205→      createdBy,
   206→    },
   207→  });
   208→}

   210→export async function POST(request: NextRequest) {
   211→  try {
   212→    // Validate request with Zod
   213→    const validationResult = validateRequest(orderCreateSchema, await request.json());
   214→
   215→    if (!validationResult.success) {
   216→      console.error('Order validation failed:', validationResult.errors);
   217→      return NextResponse.json(
   218→        {
   219→          success: false,
   220→          error: 'Invalid request data',
   221→          details: formatZodErrors(validationResult.errors)
   222→        },
   223→        { status: 400 }
   224→      );
   225→    }
   226→
   227→    const {
   228→      branchId,
   229→      cashierId,
   230→      items,
   231→      paymentMethod,
   232→      orderType,
   233→      tableId,
   234→      deliveryAddress,
   235→      deliveryAreaId,
   236→      deliveryFee,
   237→      customerId,
   238→      customerAddressId,
   239→      customerPhone,
   240→      customerName,
   241→      courierId,
   242→      loyaltyPointsRedeemed,
   243→      loyaltyDiscount,
   244→      promoCodeId,
   245→      promoDiscount,
   246→      manualDiscountPercent,
   247→      manualDiscountAmount,
   248→      manualDiscountComment,
   249→      orderNumber,
   250→      cardReferenceNumber,
   251→      paymentMethodDetail,
   252→      idempotencyKey,
   253→    } = validationResult.data;
   254→
   255→    // Check for duplicate order using idempotency key
   256→    if (idempotencyKey) {
   257→      console.log('[Order] Checking idempotency key:', idempotencyKey);
   258→
   259→      const existingKey = await db.idempotencyKey.findUnique({
   260→        where: { key: idempotencyKey },
   261→      });
   262→
   263→      if (existingKey) {
   264→        console.log('[Order] Duplicate order detected via idempotency key:', idempotencyKey);
   265→
   266→        // Return the existing order instead of creating a duplicate
   267→        const existingOrder = await db.order.findUnique({
   268→          where: { id: existingKey.key }, // Try to find by id (the key might be the order ID)
   269→        });
   270→
   271→        if (existingOrder) {
   272→          return NextResponse.json({
   273→            success: true,
   274→            order: {
   275→              id: existingOrder.id,
   276→              branchId: existingOrder.branchId,
   277→              orderNumber: existingOrder.orderNumber,
   278→              orderTimestamp: existingOrder.orderTimestamp.toISOString(),
   279→              totalAmount: existingOrder.totalAmount,
   280→              paymentMethod: existingOrder.paymentMethod,
   281→              message: 'Order already processed (idempotency)',
   282→            },
   283→            duplicate: true,
   284→          });
   285→        }
   286→
   287→        // If we have the key but can't find the order, still reject to prevent duplicates
   288→        return NextResponse.json(
   289→          {
   290→            success: false,
   291→            error: 'Duplicate order request',
   292→            message: 'This order was already processed',
   293→          },
   294→          { status: 409 } // 409 Conflict
   295→        );
   296→      }
   297→    }
   298→
   299→    // Get next order number if not provided
   300→    let finalOrderNumber = orderNumber;
   301→    if (!finalOrderNumber) {
   302→      const lastOrder = await db.order.findFirst({
   303→        where: { branchId },
   304→        orderBy: { orderNumber: 'desc' },
   305→      });
   306→      finalOrderNumber = (lastOrder?.orderNumber || 0) + 1;
   307→    }
   308→
   309→    // Get cashier info
   310→    const cashier = await db.user.findUnique({
   311→      where: { id: cashierId },
   312→    });
   313→
   314→    if (!cashier) {
   315→      return NextResponse.json(
   316→        { error: 'Cashier not found' },
   317→        { status: 404 }
   318→      );
   319→    }
   320→
   321→    // CASHIER must have their own open shift to process orders
   322→    // ADMIN and BRANCH_MANAGER can use ANY open shift from the branch
   323→    let openShift = null;
   324→
   325→    if (cashier.role === 'CASHIER') {
   326→      // Cashiers must have their own open shift
   327→      openShift = await db.shift.findFirst({
   328→        where: {
   329→          cashierId,
   330→          isClosed: false,
   331→        },
   332→      });
   333→
   334→      if (!openShift) {
   335→        return NextResponse.json(
   336→          { error: 'No active shift found. Please open a shift before processing orders.' },
   337→          { status: 400 }
   338→        );
   339→      }
   340→
   341→      // Verify that open shift is for the same branch
   342→      if (openShift.branchId !== branchId) {
   343→        return NextResponse.json(
   344→          { error: 'Active shift is for a different branch' },
   345→          { status: 400 }
   346→        );
   347→      }
   348→    } else {
   349→      // ADMIN and BRANCH_MANAGER can use any open shift from the branch
   350→      openShift = await db.shift.findFirst({
   351→        where: {
   352→          branchId,
   353→          isClosed: false,
   354→        },
   355→      });
   356→
   357→      if (!openShift) {
   358→        return NextResponse.json(
   359→          { error: `No active shift found for branch. Please have a cashier open a shift before processing orders.` },
   360→          { status: 400 }
   361→        );
   362→      }
   363→    }
   364→    
   365→    // Get branch information for tax calculation
   366→    const branch = await db.branch.findUnique({
   367→      where: { id: branchId },
   368→    });
   369→
   370→    // Calculate order totals and validate menu items
   371→    let subtotal = 0;
   372→    const orderItemsToCreate = [];
   373→    const inventoryDeductions = [];
   374→
   375→    for (const item of items) {
   376→      let menuItem;
   377→      try {
   378→        menuItem = await db.menuItem.findUnique({
   379→          where: { id: item.menuItemId },
   380→          include: {
   381→            recipes: {
   382→              include: {
   383→                ingredient: true,
   384→              },
   385→            },
   386→            categoryRel: true,
   387→            // Phase 3.1b: Include branch assignments for this branch so we can
   388→            // resolve the branch-specific price (override or global fallback).
   389→            // At most one row per branch due to @@unique([menuItemId, branchId]).
   390→            branchAssignments: {
   391→              where: { branchId },
   392→              select: { branchId: true, price: true, isActive: true },
   393→            },
   394→            ...(item.menuItemVariantId ? {
   395→              variants: {
   396→                include: {
   397→                  variantType: true,
   398→                  variantOption: true,
   399→                },
   400→                orderBy: { sortOrder: 'asc' },
   401→              },
   402→            } : {}),
   403→          },
   404→        });
   405→      } catch (error) {
   406→        console.error('[Order] Error fetching menu item:', {
   407→          menuItemId: item.menuItemId,
   408→          menuItemIdType: typeof item.menuItemId,
   409→          menuItemIdLength: item.menuItemId?.length,
   410→          error: error instanceof Error ? error.message : String(error),
   411→        });
   412→        return NextResponse.json(
   413→          {
   414→            success: false,
   415→            error: 'Invalid menu item ID format',
   416→            details: `Menu item ID "${item.menuItemId}" is not valid. Please clear your cart and try again.`
   417→          },
   418→          { status: 400 }
   419→        );
   420→      }
   421→
   422→      if (!menuItem) {
   423→        console.error('[Order] Menu item not found:', {
   424→          menuItemId: item.menuItemId,
   425→          allItemsInCart: items.map(i => ({ id: i.menuItemId })),
   426→        });
   427→        return NextResponse.json(
   428→          {
   429→            success: false,
   430→            error: 'Menu item not found',
   431→            details: `Menu item with ID "${item.menuItemId}" no longer exists. Please clear your cart and try again.`
   432→          },
   433→          { status: 404 }
   434→        );
   435→      }
   436→
   437→      if (!menuItem.isActive) {
   438→        return NextResponse.json(
   439→          {
   440→            success: false,
   441→            error: 'Menu item not available',
   442→            details: `Menu item "${menuItem.name}" is currently inactive.`
   443→          },
   444→          { status: 400 }
   445→        );
   446→      }
   447→
   448→      // Phase 3.1b: Resolve branch-specific price.
   449→      // Falls back to MenuItem.price if no override exists.
   450→      // Uses the single source of truth in src/lib/menu-pricing.ts.
   451→      const basePrice = resolveBranchPrice(
   452→        {
   453→          price: menuItem.price,
   454→          isActive: menuItem.isActive,
   455→          deletedAt: menuItem.deletedAt,
   456→          availabilityMode: menuItem.availabilityMode,
   457→          branchAssignments: menuItem.branchAssignments,
   458→        },
   459→        branchId
   460→      );
   461→
   462→      // Handle variant pricing if provided
   463→      let finalPrice = basePrice;
   464→      let variantName = null;
   465→      let variantId = null;
   466→      let customVariantValue = null;
   467→
   468→      if (item.menuItemVariantId) {
   469→        const variant = await db.menuItemVariant.findUnique({
   470→          where: { id: item.menuItemVariantId },
   471→          include: {
   472→            variantType: true,
   473→            variantOption: true,
   474→          },
   475→        });
   476→
   477→        if (variant) {
   478→          variantId = variant.id;
   479→          
   480→          // Check if this is a custom input variant
   481→          if (variant.variantType.isCustomInput && item.customVariantValue !== null && item.customVariantValue !== undefined) {
   482→            customVariantValue = parseFloat(String(item.customVariantValue));
   483→            // Ensure customVariantValue is a valid number
   484→            if (isNaN(customVariantValue) || customVariantValue <= 0) {
   485→              console.error('[Order] Invalid customVariantValue:', item.customVariantValue, 'for item:', menuItem.name);
   486→              customVariantValue = 1; // Fallback to full portion
   487→            }
   488→            // Round to 3 decimal places for clean display
   489→            const roundedValue = Math.round(customVariantValue * 1000) / 1000;
   490→            finalPrice = basePrice * customVariantValue;
   491→            variantName = `${variant.variantType.name}: ${roundedValue}x`;
   492→            console.log('[Order] Custom variant calculated:', {
   493→              itemName: menuItem.name,
   494→              basePrice: menuItem.price,
   495→              customValue: customVariantValue,
   496→              roundedValue,
   497→              finalPrice: finalPrice,
   498→              quantity: item.quantity,
   499→              subtotal: finalPrice * item.quantity
   500→            });
   501→          } else {
   502→            // Regular variant with fixed price modifier
   503→            // Phase 3.1b: modifier is applied to the branch-resolved base price
   504→            finalPrice = basePrice + variant.priceModifier;
   505→            variantName = `${variant.variantType.name}: ${variant.variantOption.name}`;
   506→          }
   507→        }
   508→      }
   509→
   510→      const itemSubtotal = finalPrice * item.quantity;
   511→      subtotal += itemSubtotal;
   512→
   513→      orderItemsToCreate.push({
   514→        menuItemId: menuItem.id,
   515→        itemName: menuItem.name,
   516→        quantity: item.quantity,
   517→        unitPrice: finalPrice,
   518→        subtotal: itemSubtotal,
   519→        recipeVersion: menuItem.version,
   520→        menuItemVariantId: variantId,
   521→        variantName,
   522→        customVariantValue,
   523→        specialInstructions: item.specialInstructions || null,
   524→        requiresCaptainReceipt: menuItem.categoryRel?.requiresCaptainReceipt || false,
   525→        // Phase 3.1b: Audit field — records that the price was resolved online
   526→        // (server-side, branch-aware). Historical orders keep NULL. Offline orders
   527→        // get 'offline' via /api/sync/batch-push.
   528→        priceSource: 'online',
   529→      });
   530→
   531→      // Calculate inventory deductions based on recipes
   532→      // Filter recipes: if variant selected, only use variant-specific recipes; otherwise use base recipes
   533→      const relevantRecipes = menuItem.recipes.filter(
   534→        recipe => recipe.menuItemVariantId === (item.menuItemVariantId || null)
   535→      );
   536→
   537→      for (const recipe of relevantRecipes) {
   538→        // Scale inventory deduction by customVariantValue if provided
   539→        const scaledQuantity = customVariantValue 
   540→          ? recipe.quantityRequired * customVariantValue 
   541→          : recipe.quantityRequired;
   542→        const totalDeduction = scaledQuantity * item.quantity;
   543→        inventoryDeductions.push({
   544→          ingredientId: recipe.ingredient.id,
   545→          ingredientName: recipe.ingredient.name,
   546→          quantityChange: -totalDeduction,
   547→          unit: recipe.unit,
   548→        });
   549→      }
   550→    }
   551→
   552→    // Calculate tax based on branch settings
   553→    const { taxAmount, taxEnabled } = calculateOrderTax(subtotal, branch);
   554→    const totalAmount = calculateTotalAmount(subtotal, taxAmount, deliveryFee || 0, loyaltyDiscount || 0, promoDiscount || 0, manualDiscountAmount || 0);
   555→
   556→    // Use the shift ID for tracking (if a shift was found)
   557→    // For CASHIER, it's their own shift; for ADMIN/BRANCH_MANAGER, it's any open shift from the branch
   558→    const currentShiftId = openShift?.id || null;
   559→
   560→    // Generate transaction hash for tamper detection
   561→    const transactionHash = Buffer.from(
   562→      `${branchId}-${finalOrderNumber}-${totalAmount}-${cashierId}-${Date.now()}`
   563→    ).toString('base64');
   564→
   565→    // Initialize ETA fields if configured for offline support
   566→    let etaFields = {};
   567→    const etaSettings = await db.branchETASettings.findUnique({
   568→      where: { branchId },
   569→    });
   570→
   571→    if (etaSettings && etaSettings.isActive) {
   572→      // Generate UUID locally for offline support
   573→      const { generateDocumentUUID } = await import('@/lib/eta/ubl-generator');
   574→      const etaUUID = generateDocumentUUID();
   575→
   576→      etaFields = {
   577→        etaUUID,
   578→        etaSubmissionStatus: 'PENDING',
   579→        etaDocumentType: '389', // 389 = Receipt
   580→        etaSettingsId: etaSettings.id,
   581→      };
   582→
   583→      console.log(`[Order] ETA initialized for offline order ${finalOrderNumber}`, {
   584→        etaUUID,
   585→        branchId,
   586→        environment: etaSettings.environment,
   587→      });
   588→    }
   589→
   590→    // Create order with inventory deduction
   591→    const order = await db.$transaction(async (tx) => {
   592→      // Create order
   593→      const newOrder = await tx.order.create({
   594→        data: {
   595→          id: `${branchId}-${finalOrderNumber}-${Date.now()}`,
   596→          branchId,
   597→          orderNumber: finalOrderNumber,
   598→          orderTimestamp: new Date(),
   599→          cashierId,
   600→          subtotal,
   601→          taxAmount,
   602→          taxEnabled,
   603→          totalAmount,
   604→          paymentMethod,
   605→          orderType: orderType || 'dine-in',
   606→          deliveryAddress: deliveryAddress || null,
   607→          deliveryAreaId: deliveryAreaId || null,
   608→          deliveryFee: deliveryFee || 0,
   609→          customerId: customerId || null,
   610→          customerAddressId: customerAddressId || null,
   611→          courierId: courierId || null,
   612→          transactionHash,
   613→          synced: false,
   614→          shiftId: currentShiftId,
   615→          tableId: tableId || null,
   616→          promoCodeId: promoCodeId || null,
   617→          promoDiscount: promoDiscount || 0,
   618→          manualDiscountPercent: manualDiscountPercent || 0,
   619→          manualDiscountAmount: manualDiscountAmount || 0,
   620→          manualDiscountComment: manualDiscountComment || null,
   621→          cardReferenceNumber: cardReferenceNumber || null,
   622→          paymentMethodDetail: paymentMethodDetail || null,
   623→          // ETA fields for offline support
   624→          ...etaFields,
   625→        },
   626→      });
   627→
   628→      // Create order items and capture created records
   629→      const createdOrderItems = [];
   630→      for (const item of orderItemsToCreate) {
   631→        const createdItem = await tx.orderItem.create({
   632→          data: {
   633→            ...item,
   634→            orderId: newOrder.id,
   635→          },
   636→        });
   637→        createdOrderItems.push(createdItem);
   638→      }
   639→
   640→      // Deduct inventory with atomic operations to prevent race conditions
   641→      for (const deduction of inventoryDeductions) {
   642→        await safeInventoryDeduct(
   643→          tx,
   644→          branchId,

   900→    if (paymentMethod === 'credit' && customerId) {

   904→        });
   905→
   906→        if (customer && (customer.customerType === 'B2B' || customer.customerType === 'BOTH')) {
   907→          const previousBalance = customer.creditBalance || 0;

  1045→    const endDate = searchParams.get('endDate');

  1047→    const offset = parseInt(searchParams.get('offset') || '0');
  1048→    const filterPaymentMethod = searchParams.get('paymentMethod');
  1049→    const filterPaymentMethodDetail = searchParams.get('paymentMethodDetail');
  1050→
  1051→    const where: any = {};
  1052→    if (branchId && branchId !== 'all') {
  1053→      where.branchId = branchId;
  1054→    }
  1055→
  1056→    // Server-side payment method filter (for Reports breakdown)
  1057→    if (filterPaymentMethod) {
  1058→      where.paymentMethod = filterPaymentMethod;
  1059→    }
  1060→    if (filterPaymentMethodDetail) {
  1061→      where.paymentMethodDetail = filterPaymentMethodDetail;
  1062→    }
  1063→
  1064→    // Filter by shiftId if provided (for shift orders view)
  1065→    if (shiftId) {
  1066→      where.shiftId = shiftId;
  1067→    }
  1068→
  1069→    if (startDate || endDate) {
  1070→      where.orderTimestamp = {};
  1071→      if (startDate) {
  1072→        where.orderTimestamp.gte = new Date(startDate);
  1073→      }
  1074→      if (endDate) {
  1075→        // Set end date to end of the day
  1076→        const endDateTime = new Date(endDate);
  1077→        endDateTime.setHours(23, 59, 59, 999);
  1078→        where.orderTimestamp.lte = endDateTime;
  1079→      }
  1080→    }

  1082→    const orders = await db.order.findMany({

  1092→        totalAmount: true,

  1094→        paymentMethod: true,
  1095→        paymentMethodDetail: true,
  1096→        orderType: true,
  1097→        deliveryFee: true,

  1099→        deliveryAreaId: true,
  1100→        isRefunded: true,
  1101→        refundReason: true,
  1102→        transactionHash: true,

  1105→        createdAt: true,

  1126→
