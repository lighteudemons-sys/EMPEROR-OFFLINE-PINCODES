    78→// Helper function to create shift offline
    79→async function createShiftOffline(shiftData: any, user: any): Promise<void> {
    80→  try {
    81→    console.log('[Shift] Creating shift offline, shiftData:', shiftData);
    82→
    83→    // Import IndexedDB storage
    84→    const { getIndexedDBStorage } = await import('@/lib/storage/indexeddb-storage');
    85→    const indexedDBStorage = getIndexedDBStorage();
    86→    console.log('[Shift] IndexedDB storage imported');
    87→
    88→    // Initialize storage if not already initialized
    89→    await indexedDBStorage.init();
    90→    console.log('[Shift] IndexedDB storage initialized');
    91→
    92→    // Get branches to find branch name
    93→    const branches = await indexedDBStorage.getAllBranches();
    94→    const branch = branches.find((b: any) => b.id === shiftData.branchId);
    95→    const branchName = branch?.branchName || 'Unknown Branch';
    96→
    97→    // Create a temporary shift ID (will be replaced on sync)
    98→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    99→    console.log('[Shift] Created tempId:', tempId);
   100→
   101→    // Create shift object with flattened cashier and branch info
   102→    const newShift = {
   103→      id: tempId,
   104→      branchId: shiftData.branchId,
   105→      branchName: branchName,
   106→      cashierId: shiftData.cashierId,
   107→      cashierName: user.name || user.username,
   108→      cashierUsername: user.username,
   109→      dayId: shiftData.dayId,
   110→      startTime: new Date().toISOString(),
   111→      openingCash: shiftData.openingCash,
   112→      openingOrders: 0,
   113→      openingRevenue: 0,
   114→      shiftNumber: 1, // Start at 1 for the day
   115→      isClosed: false,
   116→      notes: shiftData.notes,
   117→      createdAt: new Date().toISOString(),
   118→      updatedAt: new Date().toISOString(),
   119→      orderCount: 0,
   120→      currentRevenue: 0,
   121→      currentOrders: 0,
   122→    };
   123→
   124→    console.log('[Shift] Created shift object:', newShift);
   125→
   126→    // Save shift to IndexedDB
   127→    await indexedDBStorage.put('shifts', newShift);
   128→    console.log('[Shift] Shift saved to IndexedDB');
   129→
   130→    // Queue operation for sync
   131→    await indexedDBStorage.addOperation({
   132→      type: 'CREATE_SHIFT',
   133→      data: {
   134→        ...shiftData,
   135→        id: tempId,
   136→        dayId: shiftData.dayId, // Include dayId to link shift to business day
   137→        startTime: newShift.startTime,
   138→        endTime: newShift.endTime || null,
   139→        isClosed: newShift.isClosed,
   140→        createdAt: newShift.createdAt,
   141→        updatedAt: newShift.updatedAt,
   142→        cashierName: newShift.cashierName,
   143→        cashierUsername: newShift.cashierUsername,
   144→        branchName: newShift.branchName,
   145→      },
   146→      branchId: shiftData.branchId,
   147→    });
   148→    console.log('[Shift] Operation queued for sync (IndexedDB)');
   149→
   150→    console.log('[Shift] Shift created offline successfully:', newShift);
   151→  } catch (error) {
   152→    console.error('[Shift] Failed to create shift offline, error:', error);
   153→    throw error;
   154→  }
   155→}

   157→// Helper function to open business day offline
   158→async function openBusinessDayOffline(businessDayData: any, user: any): Promise<any> {
   159→  try {
   160→    console.log('[Business Day] Opening business day offline, data:', businessDayData);
   161→    console.log('[Business Day] User data:', { id: user.id, username: user.username });
   162→
   163→    const { getIndexedDBStorage } = await import('@/lib/storage/indexeddb-storage');
   164→    const indexedDBStorage = getIndexedDBStorage();
   165→    console.log('[Business Day] IndexedDB storage imported');
   166→    await indexedDBStorage.init();
   167→    console.log('[Business Day] IndexedDB storage initialized');
   168→
   169→    // Check if there's already an open business day for this branch
   170→    const existingBusinessDays = await indexedDBStorage.getBusinessDays();
   171→    console.log('[Business Day] Existing business days:', existingBusinessDays);
   172→
   173→    const existingOpenDay = existingBusinessDays.find(
   174→      (bd: any) => bd.branchId === businessDayData.branchId && bd.isOpen
   175→    );
   176→
   177→    if (existingOpenDay) {
   178→      console.log('[Business Day] Business day already open for this branch:', existingOpenDay);
   179→      return existingOpenDay;
   180→    }
   181→
   182→    // Create a temporary business day ID
   183→    const tempId = `temp-day-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
   184→    console.log('[Business Day] Creating new business day with ID:', tempId);
   185→
   186→    const newBusinessDay = {
   187→      id: tempId,
   188→      branchId: businessDayData.branchId,
   189→      openedBy: user.id,
   190→      openedAt: new Date().toISOString(),
   191→      isOpen: true,
   192→      openingCash: 0,
   193→      notes: businessDayData.notes,
   194→      totalOrders: 0,
   195→      totalSales: 0,
   196→      shifts: [],
   197→    };
   198→
   199→    console.log('[Business Day] New business day object:', newBusinessDay);
   200→
   201→    // Save business day to IndexedDB
   202→    await indexedDBStorage.saveBusinessDay(newBusinessDay);
   203→    console.log('[Business Day] Business day saved to IndexedDB with ID:', tempId);
   204→
   205→    // Verify it was saved
   206→    const verifyBusinessDays = await indexedDBStorage.getBusinessDays();
   207→    const savedDay = verifyBusinessDays.find((bd: any) => bd.id === tempId);
   208→    console.log('[Business Day] Verification - found saved day:', savedDay);
   209→
   210→    // Queue operation for sync - use IndexedDB for operations
   211→    await indexedDBStorage.addOperation({
   212→      type: 'OPEN_BUSINESS_DAY',
   213→      data: {
   214→        ...businessDayData,
   215→        id: tempId,
   216→        openedAt: newBusinessDay.openedAt,
   217→        openedBy: newBusinessDay.openedBy,
   218→      },
   219→      branchId: businessDayData.branchId,
   220→    });
   221→    console.log('[Business Day] Operation queued for sync (IndexedDB)');
   222→
   223→    console.log('[Business Day] Returning new business day:', newBusinessDay);
   224→    return newBusinessDay;
   225→  } catch (error) {
   226→    console.error('[Business Day] Failed to open business day offline, error:', error);
   227→    console.error('[Business Day] Error stack:', error instanceof Error ? error.stack : 'No stack');
   228→    throw error;
   229→  }
   230→}

   232→// Helper function to close shift offline
   233→async function closeShiftOffline(
   234→  shift: any,
   235→  closingCash: number,
   236→  notes: string,
   237→  paymentBreakdown: PaymentBreakdown
   238→): Promise<any> {
   239→  try {
   240→    console.log('[Shift] Closing shift offline, shift:', shift);
   241→
   242→    // Import IndexedDB storage
   243→    const { getIndexedDBStorage } = await import('@/lib/storage/indexeddb-storage');
   244→    const indexedDBStorage = getIndexedDBStorage();
   245→    console.log('[Shift] IndexedDB storage imported');
   246→
   247→    // Initialize storage if not already initialized
   248→    await indexedDBStorage.init();
   249→    console.log('[Shift] IndexedDB storage initialized');
   250→
   251→    // Get all orders for this shift to calculate actual revenue
   252→    // Revenue should exclude delivery fees (courier takes them)
   253→    const allOrders = await indexedDBStorage.getAllOrders();
   254→    const shiftOrders = allOrders.filter((order: any) => order.shiftId === shift.id);
   255→
   256→    console.log('[Shift] Found orders for shift:', shiftOrders.length);
   257→
   258→    // Debug: Log each order to see what fields we have
   259→    shiftOrders.forEach((order: any, idx: number) => {
   260→      console.log(`[Shift] Order ${idx + 1}:`, {
   261→        id: order.id,
   262→        orderNumber: order.orderNumber,
   263→        subtotal: order.subtotal,
   264→        totalAmount: order.totalAmount,
   265→        deliveryFee: order.deliveryFee,
   266→      });
   267→    });
   268→
   269→    // Calculate revenue excluding delivery fees
   270→    const subtotal = shiftOrders.reduce((sum: number, order: any) => sum + (order.subtotal || 0), 0);
   271→    const deliveryFees = shiftOrders.reduce((sum: number, order: any) => sum + (order.deliveryFee || 0), 0);
   272→    const cashierRevenue = subtotal; // Subtotal excludes delivery fees
   273→
   274→    // Calculate all discounts from orders
   275→    const totalLoyaltyDiscounts = shiftOrders.reduce((sum: number, order: any) => sum + (order.loyaltyDiscount || 0), 0);
   276→    const totalPromoDiscounts = shiftOrders.reduce((sum: number, order: any) => sum + (order.promoDiscount || 0), 0);
   277→    const totalManualDiscounts = shiftOrders.reduce((sum: number, order: any) => sum + (order.manualDiscountAmount || 0), 0);
   278→    const totalDiscounts = totalLoyaltyDiscounts + totalPromoDiscounts + totalManualDiscounts;
   279→
   280→    // Calculate daily expenses for this shift
   281→    const allDailyExpenses = await indexedDBStorage.getAllDailyExpenses();
   282→    const shiftDailyExpenses = allDailyExpenses.filter((exp: any) => exp.shiftId === shift.id);
   283→    const totalDailyExpenses = shiftDailyExpenses.reduce((sum: number, exp: any) => sum + (exp.amount || 0), 0);
   284→
   285→    console.log('[closeShiftOffline] Daily expenses:', {
   286→      shiftId: shift.id,
   287→      totalExpensesInDB: allDailyExpenses.length,
   288→      shiftExpensesCount: shiftDailyExpenses.length,
   289→      shiftExpenses: shiftDailyExpenses,
   290→      totalDailyExpenses,
   291→    });
   292→
   293→    console.log('[Shift] Calculated closing revenue:', {
   294→      orderCount: shiftOrders.length,
   295→      subtotal,
   296→      deliveryFees,
   297→      cashierRevenue,
   298→      totalLoyaltyDiscounts,
   299→      totalPromoDiscounts,
   300→      totalManualDiscounts,
   301→      totalDiscounts,
   302→      totalDailyExpenses,
   303→      openingCash: shift.openingCash || 0,
   304→    });
   305→
   306→    // Update shift object with calculated closing data, preserving cashier and branch info
   307→    const updatedShift = {
   308→      ...shift,
   309→      endTime: new Date().toISOString(),
   310→      closingCash: closingCash,
   311→      closingOrders: shiftOrders.length,
   312→      closingRevenue: cashierRevenue, // Use calculated revenue (subtotal only, no delivery fees)
   313→      closingLoyaltyDiscounts: totalLoyaltyDiscounts,
   314→      closingPromoDiscounts: totalPromoDiscounts,
   315→      closingDailyExpenses: totalDailyExpenses, // Use calculated daily expenses
   316→      closingVoidedItems: 0, // Would need to track voided items
   317→      closingRefunds: 0, // Would need to track refunds
   318→      isClosed: true,
   319→      notes: notes || shift.notes,
   320→      paymentBreakdown: paymentBreakdown || shift.paymentBreakdown,
   321→      updatedAt: new Date().toISOString(),
   322→      // Preserve cashier and branch info for receipt display
   323→      cashierName: shift.cashierName || shift.cashier?.name,
   324→      cashierUsername: shift.cashierUsername || shift.cashier?.username,
   325→      branchName: shift.branchName || shift.branch?.branchName,
   326→      branchId: shift.branchId,
   327→      shiftNumber: shift.shiftNumber || shift.openingOrders || shiftOrders.length,
   328→      startTime: shift.startTime || new Date().toISOString(),
   329→    };
   330→
   331→    console.log('[Shift] Updated shift object:', updatedShift);
   332→
   333→    // Save updated shift to IndexedDB
   334→    await indexedDBStorage.put('shifts', updatedShift);
   335→    console.log('[Shift] Shift updated in IndexedDB');
   336→
   337→    // Queue operation for sync
   338→    await indexedDBStorage.addOperation({
   339→      type: 'CLOSE_SHIFT',
   340→      data: {
   341→        id: shift.id,
   342→        branchId: shift.branchId,
   343→        cashierId: shift.cashierId,
   344→        startTime: shift.startTime, // Include startTime to find the shift during sync
   345→        closingCash: closingCash,
   346→        notes: notes || shift.notes,
   347→        paymentBreakdown: paymentBreakdown || shift.paymentBreakdown,
   348→        endTime: updatedShift.endTime,
   349→        // Include cashier and branch info for database creation
   350→        cashierName: updatedShift.cashierName,
   351→        cashierUsername: updatedShift.cashierUsername,
   352→        branchName: updatedShift.branchName,
   353→      },
   354→      branchId: shift.branchId,
   355→    });
   356→    console.log('[Shift] Close operation queued for sync (IndexedDB)');
   357→
   358→    console.log('[Shift] Shift closed offline successfully:', updatedShift);
   359→
   360→    // Return the updated shift for receipt generation
   361→    return updatedShift;
   362→  } catch (error) {
   363→    console.error('[Shift] Failed to close shift offline, error:', error);
   364→    throw error;
   365→  }
   366→}

   495→            const time2 = new Date(shift2.startTime).getTime();

   504→            // First, check if ID matches (exact duplicate)
   505→            if (apiShiftIds.has(offlineShift.id)) return false;
   506→
   507→            // Then, check if it's a semantic duplicate (same cashier, similar startTime)
   508→            const hasDuplicateInAPI = apiShifts.some((apiShift: any) =>
   509→              isDuplicateShift(offlineShift, apiShift)
   510→            );
   511→

   517→                duplicateWith: apiShifts.find((apiShift: any) => isDuplicateShift(offlineShift, apiShift))?.id
   518→              });
   519→              return false;
   520→            }
   521→
   522→            return true;
   523→          });

   525→          const allShifts = [...apiShifts, ...uniqueOfflineShifts];
